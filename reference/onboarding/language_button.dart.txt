import 'package:flutter/material.dart';
import 'package:shared_preferences/shared_preferences.dart';

import 'package:address/core/localization/controller/locale_controller.dart';
import 'package:address/core/localization/models/language.dart';
// ایمپورت زیر را موقتا کامنت یا حذف کردیم چون هنوز تم کامل تنظیم نشده
// import 'package:address/core/extensions/ui/theme_extensions.dart';

import 'package:address/platform/core/di/di_container.dart';

class LanguageButton extends StatefulWidget {
  const LanguageButton({super.key});

  @override
  State<LanguageButton> createState() => _LanguageButtonState();
}

class _LanguageButtonState extends State<LanguageButton> {
  late final List<Language> _languages;

  Language _selectedLanguage = Language(
    id: 2,
    code: 'fa',
    name: 'فارسی',
    flag: 'IR',
  );

  @override
  void initState() {
    super.initState();
    _languages = Language.languageList();
    _loadSelectedLanguage();
  }

  Future<void> _loadSelectedLanguage() async {
    final prefs = await SharedPreferences.getInstance();
    final languageCode = prefs.getString('selected_language_code') ?? 'fa';

    final lang = _languages.firstWhere(
      (l) => l.code == languageCode,
      orElse: () => _selectedLanguage,
    );

    if (!mounted) return;

    setState(() => _selectedLanguage = lang);

    DIContainer.get<LocaleController>().changeLocale(languageCode);
  }

  Future<void> _saveSelectedLanguage(Language language) async {
    final prefs = await SharedPreferences.getInstance();
    await prefs.setString('selected_language_code', language.code);
  }

  void _showLanguagePicker(BuildContext context) {
    // استفاده از تم استاندارد فلاتر به جای توکن‌ها
    final theme = Theme.of(context);

    showModalBottomSheet(
      context: context,
      backgroundColor: theme.colorScheme.surface,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(
          top: Radius.circular(16.0), // جایگزین tokens.radiusL
        ),
      ),
      builder: (_) {
        return Padding(
          padding: const EdgeInsets.all(16.0), // جایگزین tokens.spaceM
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text(
                "انتخاب زبان",
                style: theme.textTheme.titleMedium?.copyWith(
                  color: theme.colorScheme.onSurface,
                ),
              ),
              const Divider(),
              ..._languages.map(
                (lang) => ListTile(
                  leading: Text(
                    lang.flag,
                    style: const TextStyle(fontSize: 24),
                  ),
                  title: Text(lang.name, style: theme.textTheme.bodyLarge),
                  onTap: () async {
                    Navigator.pop(context);

                    if (!mounted) return;

                    setState(() => _selectedLanguage = lang);

                    await _saveSelectedLanguage(lang);

                    DiContainer.get<LocaleController>().changeLocale(lang.code);
                  },
                ),
              ),
            ],
          ),
        );
      },
    );
  }

  @override
  Widget build(BuildContext context) {
    // استفاده از مقادیر پیش‌فرض فلاتر
    final theme = Theme.of(context);
    final primaryColor = theme.primaryColor;
    const onPrimaryColor = Colors.white;

    return Material(
      color: primaryColor.withValues(alpha: 0.3), // جایگزین colors.primary
      borderRadius: BorderRadius.circular(22), // جایگزین tokens.radiusM
      child: InkWell(
        onTap: () => _showLanguagePicker(context),
        borderRadius: BorderRadius.circular(10.0),
        child: Padding(
          padding: const EdgeInsets.symmetric(
            horizontal: 10.0, // جایگزین tokens.spaceM
            vertical: 5.0, // جایگزین tokens.spaceS
          ),
          child: Row(
            mainAxisSize: MainAxisSize.min,
            children: [
              const Icon(Icons.language, color: onPrimaryColor),
              const SizedBox(width: 8.0),
              Text(
                _selectedLanguage.name,
                style: theme.textTheme.labelLarge?.copyWith(
                  color: onPrimaryColor,
                ),
              ),
              const SizedBox(width: 8.0),
              const Icon(Icons.keyboard_arrow_down, color: onPrimaryColor),
            ],
          ),
        ),
      ),
    );
  }
}
