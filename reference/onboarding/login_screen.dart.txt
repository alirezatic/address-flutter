import 'package:address/core/design_system/components/buttons/animated_start_button.dart';
import 'package:address/core/design_system/components/buttons/circular_rive_button.dart';
import 'package:address/core/extensions/context/build_context_extension.dart';
import 'package:address/l10n/app_localizations.dart';
import 'package:address/modules/auth/presentation/controller/login_controller.dart';
import 'package:address/modules/auth/presentation/widgets/or_divider.dart';
import 'package:country_picker/country_picker.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:address/core/extensions/string_extensions.dart';

class LoginScreen extends StatefulWidget {
  const LoginScreen({super.key, this.closeModel, this.onLoginSuccess});

  final VoidCallback? closeModel;
  final Function(String)? onLoginSuccess;

  @override
  State<LoginScreen> createState() => _LoginScreenState();
}

class _LoginScreenState extends State<LoginScreen> {
  final TextEditingController _mobileNumberController = TextEditingController();
  final LoginController _loginController = LoginController();

  String _selectedCountryCode = "IR";
  String _countryDialCode = "+98";
  OverlayEntry? _errorOverlay;

  String _emojiFlag(String cc) {
    const int flagOffset = 0x1F1E6;
    const int asciiOffset = 0x41;
    return String.fromCharCode(cc.codeUnitAt(0) - asciiOffset + flagOffset) +
        String.fromCharCode(cc.codeUnitAt(1) - asciiOffset + flagOffset);
  }

  Future<void> _handleLogin() async {
    FocusScope.of(context).unfocus();

    final String rawPhone = _mobileNumberController.text
        .toEnglishDigit()
        .trim();
    final result = await _loginController.validateAndLogin(
      rawPhone,
      _countryDialCode,
    );

    if (!mounted) return;

    if (result != null) {
      widget.onLoginSuccess?.call(result);
    } else if (_loginController.currentError != null) {
      _handleErrorMessage(_loginController.currentError!);
    }
  }

  void _handleErrorMessage(LoginError error) {
    final l10n = AppLocalizations.of(context)!;
    final message = switch (error) {
      LoginError.empty => l10n.enterMobileError,
      LoginError.notDigits => l10n.notDigits,
      LoginError.invalidPrefix => l10n.invalidPrefix,
      LoginError.invalidLength11 => l10n.enterMobileErrorWithZero,
      LoginError.invalidLength10 => l10n.enterMobileErrorWithoutZero,
      LoginError.tooShort => l10n.tooShort,
      LoginError.networkError => l10n.networkError,
    };

    _showError(message);
  }

  void _showError(String message) {
    _removeErrorOverlay();

    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;

    _errorOverlay = OverlayEntry(
      builder: (context) {
        return Positioned(
          top: context.safePadding.top + 16,
          left: 20,
          right: 20,
          child: Material(
            color: Colors.transparent,
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
              decoration: BoxDecoration(
                color: colorScheme.errorContainer,
                borderRadius: BorderRadius.circular(12),
                boxShadow: [
                  BoxShadow(
                    color: context.black.withValues(alpha: 0.15),
                    blurRadius: 15,
                    offset: const Offset(0, 6),
                  ),
                ],
              ),
              child: Row(
                children: [
                  Expanded(
                    child: Text(
                      message,
                      style: textTheme.bodyMedium?.copyWith(
                        color: colorScheme.onErrorContainer,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                  ),
                  const SizedBox(width: 12),
                  GestureDetector(
                    onTap: _removeErrorOverlay,
                    child: Icon(
                      Icons.cancel,
                      color: colorScheme.onErrorContainer,
                      size: 26,
                    ),
                  ),
                ],
              ),
            ),
          ),
        );
      },
    );

    Overlay.of(context).insert(_errorOverlay!);
    Future.delayed(const Duration(seconds: 3), _removeErrorOverlay);
  }

  void _removeErrorOverlay() {
    if (_errorOverlay != null && _errorOverlay!.mounted) {
      _errorOverlay?.remove();
      _errorOverlay = null;
    }
  }

  @override
  void dispose() {
    _removeErrorOverlay();
    _loginController.dispose();

    final tempController = _mobileNumberController;
    Future.delayed(const Duration(milliseconds: 500), tempController.dispose);

    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;
    final bottomInset = MediaQuery.viewInsetsOf(context).bottom;
    final l10n = AppLocalizations.of(context)!;

    return GestureDetector(
      onTap: () => FocusScope.of(context).unfocus(),
      child: Scaffold(
        backgroundColor: context.transparent,
        resizeToAvoidBottomInset: false,
        body: SafeArea(
          child: AnimatedContainer(
            duration: const Duration(milliseconds: 250),
            curve: Curves.fastOutSlowIn,
            padding: EdgeInsets.only(bottom: bottomInset),
            child: Center(
              child: SingleChildScrollView(
                physics: const BouncingScrollPhysics(
                  parent: AlwaysScrollableScrollPhysics(),
                ),
                child: Stack(
                  clipBehavior: Clip.none,
                  children: [
                    Container(
                      constraints: BoxConstraints(maxWidth: context.w(90)),
                      margin: EdgeInsets.all(context.w(4)),
                      padding: EdgeInsets.symmetric(
                        horizontal: context.w(5),
                        vertical: context.h(3),
                      ),
                      decoration: BoxDecoration(
                        borderRadius: BorderRadius.circular(24),
                        color: colorScheme.surface,
                        boxShadow: [
                          BoxShadow(
                            color: colorScheme.shadow.withValues(alpha: 0.5),
                            blurRadius: 20,
                            offset: const Offset(0, 8),
                          ),
                        ],
                      ),
                      child: Column(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          Text(
                            l10n.login,
                            style: textTheme.headlineMedium?.copyWith(
                              fontWeight: FontWeight.bold,
                              color: colorScheme.primary,
                            ),
                          ),
                          SizedBox(height: context.h(2)),
                          Text(
                            l10n.enteryourmobilenumber,
                            style: textTheme.titleLarge?.copyWith(
                              color: colorScheme.onSurfaceVariant,
                            ),
                            textAlign: TextAlign.center,
                          ),
                          SizedBox(height: context.h(3)),

                          _buildPhoneInputField(l10n),

                          SizedBox(height: context.h(3.5)),

                          Directionality(
                            textDirection: TextDirection.ltr,
                            child: ListenableBuilder(
                              listenable: _loginController,
                              builder: (context, child) {
                                return AnimatedStartButton(
                                  onTap: _handleLogin,
                                  isLoading: _loginController.isLoading,
                                );
                              },
                            ),
                          ),

                          SizedBox(height: context.h(3)),

                          OrDivider(),

                          SizedBox(height: context.h(3)),

                          Wrap(
                            alignment: WrapAlignment.center,
                            spacing: context.w(3),
                            runSpacing: context.h(1.5),
                            children: [
                              _buildSocialBtn(Icons.apple, "Apple"),
                              _buildSocialBtn(
                                Icons.g_mobiledata_rounded,
                                "Google",
                              ),
                              _buildSocialBtn(Icons.email_outlined, l10n.email),
                            ],
                          ),
                          SizedBox(height: context.h(1)),
                        ],
                      ),
                    ),

                    Positioned(
                      bottom: -10,
                      left: 0,
                      right: 0,
                      child: Center(
                        child: CircularRiveButton(
                          size: context.h(4.5),
                          icon: Icons.close,
                          onPressed: () {
                            _mobileNumberController.clear();
                            _removeErrorOverlay();
                            widget.closeModel?.call();
                          },
                        ),
                      ),
                    ),
                  ],
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget _buildPhoneInputField(AppLocalizations l10n) {
    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;
    final isRtl = [
      'fa',
      'ar',
    ].contains(Localizations.localeOf(context).languageCode);

    return Container(
      height: context.h(7),
      decoration: BoxDecoration(
        borderRadius: BorderRadius.circular(14),
        color: colorScheme.surface,
        boxShadow: [
          BoxShadow(
            color: colorScheme.shadow.withValues(alpha: 0.09),
            blurRadius: 10,
            offset: const Offset(0, 3),
          ),
        ],
      ),
      child: Directionality(
        textDirection: TextDirection.ltr,
        child: Row(
          crossAxisAlignment: CrossAxisAlignment.center,
          children: [
            InkWell(
              onTap: () {
                showCountryPicker(
                  context: context,
                  showPhoneCode: true,
                  onSelect: (c) {
                    setState(() {
                      _selectedCountryCode = c.countryCode;
                      _countryDialCode = "+${c.phoneCode}";
                    });
                  },
                );
              },
              borderRadius: const BorderRadius.horizontal(
                left: Radius.circular(14),
              ),
              child: Container(
                width: context.w(18),
                alignment: Alignment.center,
                child: Text(
                  "${_emojiFlag(_selectedCountryCode)} $_selectedCountryCode",
                  style: textTheme.bodyLarge,
                ),
              ),
            ),

            Container(
              height: context.h(3),
              width: 2,
              color: colorScheme.shadow.withValues(alpha: 0.2),
            ),
            Padding(
              padding: const EdgeInsets.only(left: 8, right: 2),
              child: Text(
                isRtl ? _countryDialCode.toPersianDigit() : _countryDialCode,
                style: textTheme.titleMedium?.copyWith(
                  fontWeight: FontWeight.w900,
                  fontSize: 22,
                  color: colorScheme.primary,
                ),
              ),
            ),
            Expanded(
              child: TextField(
                controller: _mobileNumberController,
                keyboardType: TextInputType.phone,
                inputFormatters: [
                  FilteringTextInputFormatter.allow(RegExp(r'[0-9۰-۹٠-٩]')),
                  if (isRtl) PersianDigitFormatter(),
                ],
                // ۱. بزرگ‌تر کردن فونت شماره تلفن
                style: textTheme.titleMedium?.copyWith(
                  letterSpacing: 3,
                  fontSize: 22,
                  fontWeight: FontWeight.w600,
                ),
                decoration: InputDecoration(
                  isDense: true,
                  // ۳. حذف رنگ پس‌زمینه فیلد متنی (حذف رنگ خاکستری)
                  filled: false,
                  hintText: l10n.mobileNumber,
                  hintStyle: textTheme.titleMedium?.copyWith(
                    color: colorScheme.shadow.withValues(alpha: 0.3),
                    fontSize: 20,
                  ),
                  // ۲. حذف کامل حاشیه (Border)
                  border: InputBorder.none,
                  enabledBorder: InputBorder.none,
                  focusedBorder: InputBorder.none,
                  contentPadding: const EdgeInsets.symmetric(horizontal: 4),
                ),
              ),
            ),
          ],
        ),
      ),
    );
  }

  Widget _buildSocialBtn(IconData icon, String label) {
    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;

    return InkWell(
      onTap: () {},
      borderRadius: BorderRadius.circular(14),
      child: Container(
        padding: EdgeInsets.symmetric(
          horizontal: context.w(4),
          vertical: context.h(1.2),
        ),
        decoration: BoxDecoration(
          color: colorScheme.surface, // همرنگ پس‌زمینه کارت
          borderRadius: BorderRadius.circular(14),
          boxShadow: [
            BoxShadow(
              color: colorScheme.shadow.withValues(
                alpha: 0.4,
              ), // اضافه شدن سایه
              blurRadius: 8,
              offset: const Offset(0, 3),
            ),
          ],
        ),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            Icon(icon, color: colorScheme.onSurface, size: 20),
            const SizedBox(width: 8),
            Text(
              label,
              style: textTheme.labelLarge?.copyWith(
                fontWeight: FontWeight.w600,
                color: colorScheme.primary,
              ),
            ),
          ],
        ),
      ),
    );
  }
}
