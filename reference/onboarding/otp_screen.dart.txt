import 'dart:async';
import 'package:address/core/design_system/components/buttons/animated_start_button.dart';
import 'package:address/core/design_system/components/buttons/circular_rive_button.dart';
import 'package:address/core/extensions/context/build_context_extension.dart';
import 'package:address/l10n/app_localizations.dart';
import 'package:address/modules/home/presentation/screens/home.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:pin_code_fields/pin_code_fields.dart';
import 'package:address/core/extensions/string_extensions.dart';

class OtpScreen extends StatefulWidget {
  final String phoneNumber;
  final VoidCallback onBack;

  const OtpScreen({super.key, required this.phoneNumber, required this.onBack});

  @override
  State<OtpScreen> createState() => _OtpScreenState();
}

class _OtpScreenState extends State<OtpScreen> {
  int num = 60;
  Timer? _timer;

  final TextEditingController otpController = TextEditingController();
  final StreamController<ErrorAnimationType> errorController =
      StreamController<ErrorAnimationType>();

  final OtpController _otpController = OtpController();
  OverlayEntry? _errorOverlay;

  bool _isLoading = false;

  @override
  void initState() {
    super.initState();
    startTimer();
  }

  void startTimer() {
    _timer?.cancel();
    _timer = Timer.periodic(const Duration(seconds: 1), (timer) {
      if (num == 0) {
        timer.cancel();
        if (mounted) setState(() {});
      } else {
        if (mounted) {
          setState(() {
            num--;
          });
        }
      }
    });
  }

  void resendCode() {
    if (num == 0) {
      setState(() {
        num = 60;
      });
      otpController.clear();
      startTimer();
    }
  }

  void _handleBack() {
    otpController.clear();
    _removeErrorOverlay();
    widget.onBack();
  }

  void _showError(String message) {
    _removeErrorOverlay();

    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;

    _errorOverlay = OverlayEntry(
      builder: (context) {
        return Positioned(
          top: context.safePadding.top + 16,
          left: 20,
          right: 20,
          child: Material(
            color: Colors.transparent,
            child: Container(
              padding: const EdgeInsets.symmetric(horizontal: 16, vertical: 14),
              decoration: BoxDecoration(
                color: colorScheme.errorContainer,
                borderRadius: BorderRadius.circular(12),
                boxShadow: [
                  BoxShadow(
                    color: context.black.withValues(alpha: 0.2),
                    blurRadius: 10,
                    offset: const Offset(0, 4),
                  ),
                ],
              ),
              child: Row(
                children: [
                  Expanded(
                    child: Text(
                      message,
                      style: textTheme.bodyMedium?.copyWith(
                        color: colorScheme.onErrorContainer,
                        fontWeight: FontWeight.w600,
                      ),
                    ),
                  ),
                  const SizedBox(width: 12),
                  GestureDetector(
                    onTap: _removeErrorOverlay,
                    child: Icon(
                      Icons.cancel,
                      color: colorScheme.onErrorContainer,
                      size: 26,
                    ),
                  ),
                ],
              ),
            ),
          ),
        );
      },
    );

    Overlay.of(context).insert(_errorOverlay!);
    Future.delayed(const Duration(seconds: 3), _removeErrorOverlay);
  }

  void _removeErrorOverlay() {
    if (_errorOverlay != null && _errorOverlay!.mounted) {
      _errorOverlay?.remove();
      _errorOverlay = null;
    }
  }

  Future<void> _verifyOtp() async {
    FocusScope.of(context).unfocus();

    if (otpController.text.length != 5) {
      errorController.add(ErrorAnimationType.shake);
      final l10n = AppLocalizations.of(context)!;
      _showError(l10n.enterFullOtpCode);
      return;
    }

    if (_isLoading) return;

    setState(() {
      _isLoading = true;
    });

    String englishCode = otpController.text;
    const persian = ['۰', '۱', '۲', '۳', '۴', '۵', '۶', '۷', '۸', '۹'];
    const arabic = ['٠', '١', '٢', '٣', '٤', '٥', '٦', '٧', '٨', '٩'];
    for (int i = 0; i < 10; i++) {
      englishCode = englishCode
          .replaceAll(persian[i], i.toString())
          .replaceAll(arabic[i], i.toString());
    }

    final errorResult = await _otpController.verifyOtp(
      widget.phoneNumber,
      englishCode,
    );

    if (!context.mounted) return;

    setState(() {
      _isLoading = false;
    });

    if (errorResult == OtpError.none) {
      _removeErrorOverlay();
      Navigator.pushReplacement(
        context,
        MaterialPageRoute(builder: (context) => const Home()),
      );
    } else {
      errorController.add(ErrorAnimationType.shake);
      otpController.clear();

      final l10n = AppLocalizations.of(context)!;
      if (errorResult == OtpError.networkError) {
        _showError(l10n.networkError);
      } else {
        _showError('کد وارد شده صحیح نمی‌باشد');
      }
    }
  }

  @override
  void dispose() {
    _timer?.cancel();
    _removeErrorOverlay();
    errorController.close();

    final tempController = otpController;
    Future.delayed(const Duration(milliseconds: 500), tempController.dispose);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    final l10n = AppLocalizations.of(context)!;
    final appDirection = Directionality.of(context);

    final isRtl = [
      'fa',
      'ar',
    ].contains(Localizations.localeOf(context).languageCode);
    final displayPhone = isRtl
        ? widget.phoneNumber.toPersianDigit()
        : widget.phoneNumber;
    final displayTimer = isRtl
        ? num.toString().toPersianDigit()
        : num.toString();

    final colorScheme = context.colorScheme;
    final textTheme = context.textTheme;
    final tokens = context.tokens;
    final bottomInset = MediaQuery.viewInsetsOf(context).bottom;

    return GestureDetector(
      onTap: () => FocusScope.of(context).unfocus(),
      child: Scaffold(
        backgroundColor: context.transparent,
        resizeToAvoidBottomInset: false,
        body: SafeArea(
          child: Directionality(
            textDirection: TextDirection.ltr,
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 300),
              curve: Curves.easeOut,
              padding: EdgeInsets.only(bottom: bottomInset),
              child: Center(
                child: SingleChildScrollView(
                  physics: const BouncingScrollPhysics(),
                  child: Stack(
                    clipBehavior: Clip.none,
                    children: [
                      Container(
                        constraints: BoxConstraints(maxWidth: context.w(90)),
                        margin: EdgeInsets.all(tokens.spaceM),
                        padding: const EdgeInsets.all(1.0),
                        decoration: BoxDecoration(
                          borderRadius: BorderRadius.circular(tokens.radiusXL),
                          gradient: LinearGradient(
                            colors: [
                              colorScheme.surface.withValues(alpha: 0.9),
                              colorScheme.surfaceContainerHighest,
                              colorScheme.surfaceContainer.withValues(
                                alpha: 0.5,
                              ),
                            ],
                          ),
                        ),
                        child: Container(
                          padding: EdgeInsets.symmetric(
                            horizontal: tokens.spaceL,
                            vertical: tokens.spaceXL,
                          ),
                          decoration: BoxDecoration(
                            borderRadius: BorderRadius.circular(
                              tokens.radiusXL,
                            ),
                            color: colorScheme.surface,
                            boxShadow: [
                              BoxShadow(
                                color: colorScheme.shadow.withValues(
                                  alpha: 0.1,
                                ),
                                offset: const Offset(0, 4),
                                blurRadius: 8,
                              ),
                            ],
                          ),
                          child: Column(
                            mainAxisSize: MainAxisSize.min,
                            children: [
                              Text(
                                l10n.welcome,
                                style: textTheme.titleLarge?.copyWith(
                                  fontWeight: FontWeight.bold,
                                  color: colorScheme.onSurface,
                                ),
                              ),
                              SizedBox(height: tokens.spaceS),
                              Divider(
                                color: colorScheme.outlineVariant.withValues(
                                  alpha: 0.5,
                                ),
                              ),
                              SizedBox(height: tokens.spaceM),
                              Text(
                                l10n.enterOtpCode('\u200E$displayPhone\u200E'),
                                textAlign: TextAlign.center,
                                textDirection: appDirection,
                                style: textTheme.bodyMedium?.copyWith(
                                  color: colorScheme.onSurfaceVariant,
                                ),
                              ),
                              SizedBox(height: tokens.spaceXL),
                              PinCodeTextField(
                                appContext: context,
                                length: 5,
                                obscureText: false,
                                animationType: AnimationType.fade,
                                keyboardType: TextInputType.number,
                                inputFormatters: [
                                  FilteringTextInputFormatter.allow(
                                    RegExp(r'[0-9۰-۹٠-٩]'),
                                  ),
                                  if (isRtl) PersianDigitFormatter(),
                                ],
                                textStyle: textTheme.titleLarge?.copyWith(
                                  fontWeight: FontWeight.bold,
                                  color: colorScheme.onSurface,
                                ),
                                pinTheme: PinTheme(
                                  shape: PinCodeFieldShape.box,
                                  borderRadius: BorderRadius.circular(
                                    tokens.radiusM,
                                  ),
                                  fieldHeight: 50,
                                  fieldWidth: 50,
                                  activeFillColor: colorScheme.surface,
                                  inactiveColor: colorScheme.outlineVariant,
                                  inactiveFillColor: colorScheme
                                      .surfaceContainerHighest
                                      .withValues(alpha: 0.5),
                                  selectedFillColor: colorScheme.surface,
                                  selectedColor: colorScheme.primary,
                                  activeColor: colorScheme.primary,
                                ),
                                animationDuration: const Duration(
                                  milliseconds: 300,
                                ),
                                backgroundColor: Colors.transparent,
                                enableActiveFill: true,
                                errorAnimationController: errorController,
                                controller: otpController,
                                onCompleted: (value) => _verifyOtp(),
                                onChanged: (value) {},
                              ),
                              SizedBox(height: tokens.spaceL),
                              InkWell(
                                onTap: num == 0 ? resendCode : null,
                                borderRadius: BorderRadius.circular(50.0),
                                child: Container(
                                  decoration: BoxDecoration(
                                    borderRadius: BorderRadius.circular(50.0),
                                    color: colorScheme.surfaceContainerHighest
                                        .withValues(alpha: 0.5),
                                  ),
                                  padding: EdgeInsets.symmetric(
                                    horizontal: tokens.spaceL,
                                    vertical: tokens.spaceS,
                                  ),
                                  child: Text(
                                    num > 0
                                        ? l10n.didNotReceiveCode(displayTimer)
                                        : l10n.resendCode,
                                    style: textTheme.labelLarge?.copyWith(
                                      fontWeight: FontWeight.bold,
                                      color: num > 0
                                          ? colorScheme.onSurfaceVariant
                                          : colorScheme.primary,
                                    ),
                                  ),
                                ),
                              ),
                              SizedBox(height: tokens.spaceXL),
                              AnimatedStartButton(
                                onTap: _verifyOtp,
                                isLoading: _isLoading,
                              ),
                              SizedBox(height: tokens.spaceL),
                              Divider(
                                color: colorScheme.outlineVariant.withValues(
                                  alpha: 0.5,
                                ),
                              ),
                              SizedBox(height: tokens.spaceM),
                              InkWell(
                                onTap: _handleBack,
                                child: Text(
                                  l10n.changeNumber,
                                  style: textTheme.bodyMedium?.copyWith(
                                    fontWeight: FontWeight.bold,
                                    color: colorScheme.primary,
                                  ),
                                ),
                              ),
                              SizedBox(height: context.h(2)),
                            ],
                          ),
                        ),
                      ),

                      Positioned(
                        bottom: -5,
                        left: 0,
                        right: 0,
                        child: Center(
                          child: CircularRiveButton(
                            size: context.h(4),
                            icon: Icons.arrow_back_rounded,
                            onPressed: _handleBack,
                          ),
                        ),
                      ),
                    ],
                  ),
                ),
              ),
            ),
          ),
        ),
      ),
    );
  }
}

// ---------------------------------------------------------
// OtpController
// ---------------------------------------------------------

enum OtpError { none, empty, tooShort, networkError, unknown }

class OtpController extends ChangeNotifier {
  bool _isLoading = false;
  bool get isLoading => _isLoading;

  OtpError _error = OtpError.none;
  OtpError get error => _error;

  bool _isSuccess = false;
  bool get isSuccess => _isSuccess;

  Future<OtpError> verifyOtp(String phone, String code) async {
    if (code.isEmpty) return OtpError.empty;
    if (code.length < 5) return OtpError.tooShort;

    _isLoading = true;
    _error = OtpError.none;
    notifyListeners();

    try {
      await Future.delayed(const Duration(seconds: 1));

      _isSuccess = true;
      _isLoading = false;
      notifyListeners();
      return OtpError.none;
    } catch (e) {
      _error = OtpError.networkError;
      _isLoading = false;
      notifyListeners();
      return _error;
    }
  }
}
