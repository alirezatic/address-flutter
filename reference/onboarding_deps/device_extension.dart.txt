import 'package:address/core/design_system/responsive/breakpoints.dart';
import 'package:flutter/material.dart';

extension DeviceContextExtension on BuildContext {
  double get width => MediaQuery.of(this).size.width;

  double get height => MediaQuery.of(this).size.height;

  DeviceType get device => AppBreakpoints.getDevice(width);

  bool get isMobile => device == DeviceType.mobile;

  bool get isTablet => device == DeviceType.tablet;

  bool get isDesktop => device == DeviceType.desktop;

  bool get isLargeDesktop => device == DeviceType.largeDesktop;
}
