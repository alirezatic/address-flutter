import 'dart:ui';
import 'package:address/core/design_system/components/buttons/animated_start_button.dart';
import 'package:address/core/design_system/components/buttons/language_button.dart';
import 'package:address/core/extensions/context/build_context_extension.dart';
import 'package:address/modules/auth/presentation/screens/login_screen.dart';
import 'package:address/modules/auth/presentation/screens/otp_screen.dart';
import 'package:flutter/material.dart';
import 'package:flutter/physics.dart';
import 'package:rive/rive.dart' as rive;

class OnboardingView extends StatefulWidget {
  const OnboardingView({super.key});

  @override
  State<OnboardingView> createState() => _OnboardingViewState();
}

class _OnboardingViewState extends State<OnboardingView>
    with TickerProviderStateMixin {
  late final AnimationController _logInAnimController;
  late final Animation<double> _smoothAnim;
  late final rive.RiveAnimationController _btnController;

  late final AnimationController _otpAnimController;
  late final Animation<Offset> _otpSlideAnimation;

  String? _enteredPhoneNumber;

  @override
  void initState() {
    super.initState();

    _logInAnimController = AnimationController(
      duration: const Duration(milliseconds: 650),
      upperBound: 1,
      vsync: this,
    );

    _smoothAnim = CurvedAnimation(
      parent: _logInAnimController,
      curve: Curves.easeOutCubic,
      reverseCurve: Curves.easeInCubic,
    );

    _btnController = rive.OneShotAnimation("active", autoplay: false);

    const springDesc = SpringDescription(
      mass: 0.25,
      stiffness: 18,
      damping: 12,
    );

    _btnController.isActiveChanged.addListener(() {
      if (!_btnController.isActive) {
        final springAnim = SpringSimulation(springDesc, 0, 1, 0);
        _logInAnimController.animateWith(springAnim);
      }
    });

    _otpAnimController = AnimationController(
      duration: const Duration(milliseconds: 500),
      vsync: this,
    );

    _otpSlideAnimation =
        Tween<Offset>(begin: const Offset(0, 1.2), end: Offset.zero).animate(
          CurvedAnimation(
            parent: _otpAnimController,
            curve: Curves.easeOutQuart,
            reverseCurve: Curves.easeInQuart,
          ),
        );
  }

  @override
  void dispose() {
    _logInAnimController.dispose();
    _otpAnimController.dispose();
    _btnController.dispose();
    super.dispose();
  }

  void _goToOtpScreen(String phoneNumber) {
    setState(() {
      _enteredPhoneNumber = phoneNumber;
    });

    _logInAnimController.reverse();

    Future.delayed(const Duration(milliseconds: 300), () {
      if (mounted) _otpAnimController.forward();
    });
  }

  void _backToLoginScreen() {
    _otpAnimController.reverse();

    Future.delayed(const Duration(milliseconds: 300), () {
      if (mounted) _logInAnimController.forward();
    });
  }

  @override
  Widget build(BuildContext context) {
    final height = MediaQuery.of(context).size.height;
    // گرفتن رنگ‌های تم جاری
    final colorScheme = Theme.of(context).colorScheme;

    return Scaffold(
      resizeToAvoidBottomInset: false,
      // استفاده از رنگ بک‌گراند تم به جای پیش‌فرض
      backgroundColor: colorScheme.surface,
      body: Stack(
        children: [
          AnimatedBuilder(
            animation: _smoothAnim,
            builder: (context, child) {
              final scale = 1 - (_smoothAnim.value * 0.03);
              return Transform.scale(scale: scale, child: child);
            },
            child: RepaintBoundary(
              child: Stack(
                children: [
                  ImageFiltered(
                    imageFilter: ImageFilter.blur(sigmaX: 40, sigmaY: 40),
                    child: Center(
                      child: OverflowBox(
                        maxWidth: double.infinity,
                        child: Transform.translate(
                          offset: const Offset(200, 100),
                          child: Image.asset(
                            "assets/images/background/spline.png",
                            width: context.h(71.5),
                            height: context.h(56),
                            fit: BoxFit.cover,
                          ),
                        ),
                      ),
                    ),
                  ),
                  ImageFiltered(
                    imageFilter: ImageFilter.blur(sigmaX: 50, sigmaY: 50),
                    child: const rive.RiveAnimation.asset(
                      "assets/rive/shapes.riv",
                    ),
                  ),
                ],
              ),
            ),
          ),

          AnimatedBuilder(
            animation: _smoothAnim,
            builder: (context, child) {
              return Transform.translate(
                offset: Offset(0, -30 * _smoothAnim.value),
                child: child,
              );
            },
            child: RepaintBoundary(
              child: SafeArea(
                child: Padding(
                  padding: EdgeInsets.fromLTRB(
                    context.h(4),
                    context.h(4),
                    context.h(4),
                    context.h(12),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.center,
                    children: [
                      Image.asset(
                        "assets/images/logo/address.png",
                        height: context.h(28),
                      ),
                      const Spacer(),

                      Center(
                        child: Directionality(
                          textDirection: TextDirection.ltr,
                          child: AnimatedStartButton(
                            isLoading: false,
                            onTap: () {
                              _logInAnimController.forward();
                            },
                          ),
                        ),
                      ),

                      SizedBox(height: context.h(6)),

                      const LanguageButton(),
                    ],
                  ),
                ),
              ),
            ),
          ),

          // لایه سایه تیره (Overlay)
          AnimatedBuilder(
            animation: Listenable.merge([_smoothAnim, _otpAnimController]),
            builder: (context, child) {
              final opacityVal = (_smoothAnim.value > _otpAnimController.value)
                  ? _smoothAnim.value
                  : _otpAnimController.value;
              return IgnorePointer(
                ignoring: true,
                child: Opacity(
                  opacity: 0.55 * opacityVal,
                  // 👈 جایگزینی AppColors.black با رنگ استاندارد تم
                  child: Container(color: colorScheme.scrim),
                ),
              );
            },
          ),

          AnimatedBuilder(
            animation: _smoothAnim,
            builder: (context, child) {
              final curveValue = Curves.easeOutCubic.transform(
                _smoothAnim.value,
              );
              final yOffset = -height * (1 - curveValue);

              final isHidden = _smoothAnim.value == 0;

              return Transform.translate(
                offset: Offset(0, yOffset),
                child: Transform.scale(
                  scale: 0.98 + (curveValue * 0.02),
                  child: ExcludeFocus(excluding: isHidden, child: child!),
                ),
              );
            },
            child: RepaintBoundary(
              child: LoginScreen(
                closeModel: () => _logInAnimController.reverse(),
                onLoginSuccess: _goToOtpScreen,
              ),
            ),
          ),

          SlideTransition(
            position: _otpSlideAnimation,
            child: _enteredPhoneNumber != null
                ? RepaintBoundary(
                    child: OtpScreen(
                      phoneNumber: _enteredPhoneNumber!,
                      onBack: _backToLoginScreen,
                    ),
                  )
                : const SizedBox.shrink(),
          ),
        ],
      ),
    );
  }
}
